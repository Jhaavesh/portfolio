// MCP Sales Analytics Integration API
[ApiController]
[Route("api/v1/mcp/analytics")]
[Authorize(Roles = "AI_Agent")]
public class SalesAnalyticsController : ControllerBase
{
    private readonly ISalesService _salesService;

    [HttpPost("get_sales_stats")]
    public async Task<IActionResult> GetStats([FromBody] McpQuery query)
    {
        var kpiResult = await _salesService
            .GetStoreKpiAsync(query.StoreId, query.DateRange);
        
        return Ok(new { Status = "Success", Data = kpiResult });
    }
}

// Clean Architecture Dependency Injection Setup
public static class DependencyInjection
{
    public static IServiceCollection AddInfrastructure(this IServiceCollection services, IConfiguration config)
    {
        services.AddDbContext<ApplicationDbContext>(options =>
            options.UseSqlServer(config.GetConnectionString("DefaultConnection")));

        services.AddScoped<IJwtTokenGenerator, JwtTokenGenerator>();
        services.AddScoped<IMcpAnalyticsService, McpAnalyticsService>();

        return services;
    }
}
